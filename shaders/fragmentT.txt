#version 330 core

in vec3 vColor;
in vec3 Normal;
in vec3 fragPos;

out vec4 FragColor;

uniform vec3 camera_pos;
uniform float alpha;

float diffuseLight(){
	vec3 lightDir = normalize(camera_pos-fragPos);
	float kd = 0.8;
	float result = kd * max(dot(normalize(Normal),lightDir),0.0);
	//float result = 0.0;
	return result;
}

void main() {
	float ambient = 0.15;
	float diffuse = diffuseLight();
    FragColor = vec4((ambient + diffuse) * vColor, alpha);
    //FragColor = vec4(normalize(Normal) * 0.5 + 0.5, 1.0);
}
